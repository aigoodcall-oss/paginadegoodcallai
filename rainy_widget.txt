<script>
(function(){
  document.getElementById('voz-fab').classList.add('visible');

  var ws = null;
  var micCtx = null;   // AudioContext para captura (16kHz)
  var playCtx = null;  // AudioContext para reproducción (24kHz)
  var processor = null;
  var source = null;
  var stream = null;
  var activo = false;
  var nextPlayAt = 0;  // scheduling secuencial — igual que jitter buffer del .52
  var activeSources = [];  // buffers sonando/programados — para poder cortarlos en una interrupcion
  var PARTIDA_ID = 16;
  var WS_URL = 'wss://crm.goodcallai.net/ws/voz-live?partidaId=' + PARTIDA_ID;
  var CHAT_URL = 'https://crm.goodcallai.net/api/rainy/chat-web';
  var vozHistorial = [
    { role: 'assistant', content: '¡Hola! Soy Rainy 👋 ¿En qué puedo ayudarte hoy?' }
  ];
  var vozEnviandoTexto = false;
  var estadoTimer = null;
  var vozSonidos = {};
  function vozReproducirSonido(nombre){
    try {
      var snd = vozSonidos[nombre];
      if(!snd){ snd = new Audio('sounds/rainy-' + nombre + '.wav'); snd.volume = 0.45; vozSonidos[nombre] = snd; }
      snd.currentTime = 0;
      snd.play().catch(function(){});
    } catch(ex){}
  }

  // Estado normal (persiste hasta el proximo setEstado)
  function setEstado(txt){
    if(estadoTimer){ clearTimeout(estadoTimer); estadoTimer = null; }
    var el = document.getElementById('voz-estado');
    el.style.opacity = '1';
    el.textContent = txt;
  }
  // Estado de transicion: aparece, y luego de "ms" se desvanece y se vacia ("se quita")
  function setEstadoTemporal(txt, ms){
    setEstado(txt);
    estadoTimer = setTimeout(function(){
      var el = document.getElementById('voz-estado');
      el.style.opacity = '0';
      estadoTimer = setTimeout(function(){ el.textContent = ''; }, 400);
    }, ms);
  }

  // Orbe flotante: se activa la animacion "hablando" mientras haya audio de la IA sonando
  function vozOrbHablando(on){
    var wrap = document.getElementById('voz-orb-wrap');
    var orb = document.getElementById('voz-orb');
    if(!wrap || !orb) return;
    wrap.classList.toggle('hablando', !!on);
    orb.classList.toggle('hablando', !!on);
  }

  // Corta de inmediato todo lo que suena o esta programado (igual que _fadeOutAndStop del .52)
  function detenerReproduccionActual(){
    activeSources.forEach(function(s){ try { s.stop(); s.disconnect(); } catch(ex){} });
    activeSources = [];
    nextPlayAt = 0;
    vozOrbHablando(false);
  }

  function resetPlayQueue(){ detenerReproduccionActual(); }

  // Reproduce PCM 16-bit encadenando chunks sin gaps ni explosiones
  function encolarPCM(b64, sampleRate){
    try {
      if(!playCtx) return;
      var sr = sampleRate || 24000;
      var bin = atob(b64);
      var bytes = new Uint8Array(bin.length);
      for(var i=0;i<bin.length;i++) bytes[i]=bin.charCodeAt(i);
      var int16 = new Int16Array(bytes.buffer);
      var float32 = new Float32Array(int16.length);
      for(var j=0;j<int16.length;j++) float32[j] = int16[j] / 32768.0;
      var buf = playCtx.createBuffer(1, float32.length, sr);
      buf.getChannelData(0).set(float32);
      var src = playCtx.createBufferSource();
      src.buffer = buf;
      src.connect(playCtx.destination);
      // Encadenar secuencialmente, sin recortes: igual que el jitter buffer del .52, dejamos que la
      // cola crezca y se reproduzca a ritmo real — cortar a mitad de un buffer suena entrecortado y
      // se come palabras. Lo unico que debe vaciar la cola es una interrupcion real (interrumpido).
      var now = playCtx.currentTime;
      var startAt = Math.max(now + 0.05, nextPlayAt);
      src.start(startAt);
      activeSources.push(src);
      vozOrbHablando(true);
      src.onended = function(){
        var i = activeSources.indexOf(src);
        if(i >= 0) activeSources.splice(i, 1);
        if(activeSources.length === 0) vozOrbHablando(false);
      };
      nextPlayAt = startAt + buf.duration;
    } catch(ex){ console.warn('PCM error', ex); }
  }

  // ── Chat de texto con Rainy ──
  function vozAgregarMensaje(rol, texto){
    var cont = document.getElementById('voz-chat-msgs');
    var div = document.createElement('div');
    div.className = 'voz-msg ' + (rol === 'user' ? 'user' : 'bot');
    div.textContent = texto;
    cont.appendChild(div);
    cont.scrollTop = cont.scrollHeight;
  }

  function vozMostrarEscribiendo(){
    var cont = document.getElementById('voz-chat-msgs');
    var div = document.createElement('div');
    div.className = 'voz-msg bot cargando';
    div.id = 'voz-escribiendo';
    div.innerHTML = '<span></span><span></span><span></span>';
    cont.appendChild(div);
    cont.scrollTop = cont.scrollHeight;
  }
  function vozQuitarEscribiendo(){
    var el = document.getElementById('voz-escribiendo');
    if(el) el.remove();
  }

  window.vozEnviarTexto = function(){
    if(vozEnviandoTexto) return;
    var input = document.getElementById('voz-chat-input');
    var texto = (input.value || '').trim();
    if(!texto) return;

    vozAgregarMensaje('user', texto);
    vozHistorial.push({ role: 'user', content: texto });
    input.value = '';
    input.style.height = 'auto';
    document.getElementById('voz-chat-input-wrap').classList.remove('tiene-texto');
    vozEnviandoTexto = true;
    vozMostrarEscribiendo();

    fetch(CHAT_URL, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ partidaId: PARTIDA_ID, messages: vozHistorial })
    }).then(function(r){ return r.json(); }).then(function(data){
      vozQuitarEscribiendo();
      vozEnviandoTexto = false;
      var reply = (data && data.reply) ? data.reply : 'Disculpa, no pude responder justo ahora. ¿Puedes intentar de nuevo en un momento?';
      vozAgregarMensaje('bot', reply);
      vozHistorial.push({ role: 'assistant', content: reply });
    }).catch(function(){
      vozQuitarEscribiendo();
      vozEnviandoTexto = false;
      vozAgregarMensaje('bot', 'Hubo un problema de conexión. Intenta de nuevo en un momento 🙏');
    });
  };

  // Boton derecho del chat: si hay texto escrito, envia; si esta vacio, abre el modo de voz (como GPT)
  window.vozAccionPrincipal = function(){
    var ta = document.getElementById('voz-chat-input');
    if(ta && ta.value.trim()) window.vozEnviarTexto();
    else window.vozAbrirVoz();
  };

  // Boton de dictado: convierte lo que dices en texto dentro del chat (no abre el modo de voz)
  var vozReco = null;
  var vozDictando = false;
  window.vozToggleDictado = function(){
    var SR = window.SpeechRecognition || window.webkitSpeechRecognition;
    var btn = document.getElementById('voz-btn-dictar');
    if(!SR){ alert('Tu navegador no soporta el dictado por voz.'); return; }
    if(vozDictando){ if(vozReco) vozReco.stop(); return; }

    var ta = document.getElementById('voz-chat-input');
    var base = (ta.value || '').trim();
    vozReco = new SR();
    vozReco.lang = 'es-ES';
    vozReco.continuous = true;
    vozReco.interimResults = true;

    vozReco.onresult = function(e){
      var texto = '';
      for(var i = 0; i < e.results.length; i++) texto += e.results[i][0].transcript;
      ta.value = (base ? base + ' ' : '') + texto;
      ta.dispatchEvent(new Event('input'));
    };
    vozReco.onend = function(){
      vozDictando = false;
      if(btn) btn.classList.remove('activo');
    };
    vozReco.onerror = function(){
      vozDictando = false;
      if(btn) btn.classList.remove('activo');
    };

    try {
      vozReco.start();
      vozDictando = true;
      if(btn) btn.classList.add('activo');
    } catch(ex){}
  };

  // Enter envia, Shift+Enter hace salto de linea; el textarea crece con el contenido
  (function(){
    var ta = document.getElementById('voz-chat-input');
    if(!ta) return;
    ta.addEventListener('input', function(){
      ta.style.height = 'auto';
      ta.style.height = Math.min(ta.scrollHeight, 84) + 'px';
      var wrap = document.getElementById('voz-chat-input-wrap');
      if(ta.value.trim()) wrap.classList.add('tiene-texto');
      else wrap.classList.remove('tiene-texto');
    });
    ta.addEventListener('keydown', function(e){
      if(e.key === 'Enter' && !e.shiftKey){
        e.preventDefault();
        window.vozEnviarTexto();
      }
    });
  })();

  // ── Apertura del panel: por defecto siempre cae en el chat de texto ──
  function vozMostrarVistaChat(){
    document.getElementById('voz-vista-voz').classList.remove('activa');
    document.getElementById('voz-vista-chat').classList.remove('oculta');
  }
  function vozMostrarVistaVoz(){
    document.getElementById('voz-vista-chat').classList.add('oculta');
    document.getElementById('voz-vista-voz').classList.add('activa');
  }

  window.vozCerrarPanel = function(){
    document.getElementById('voz-panel').classList.remove('open');
    document.body.classList.remove('rainy-abierta');
    if(activo) vozDetener();
    vozMostrarVistaChat();
  };

  window.vozToggle = function(){
    var panel = document.getElementById('voz-panel');
    if(!panel.classList.contains('open')){
      panel.classList.add('open');
      document.body.classList.add('rainy-abierta');
      if(typeof window.closeBubble === 'function') window.closeBubble();
    } else {
      if(activo) vozDetener();
      panel.classList.remove('open');
      document.body.classList.remove('rainy-abierta');
      vozMostrarVistaChat();
    }
  };

  // El icono de microfono (arriba-derecha del textarea) es la unica puerta a la voz
  window.vozAbrirVoz = function(){
    vozMostrarVistaVoz();
    vozIniciar();
  };

  window.vozVolverChat = function(){
    if(activo) vozDetener();
    vozMostrarVistaChat();
  };

  function vozIniciar(){
    if(activo) return;
    activo = true;
    setEstado('Cargando Rainy...');
    document.getElementById('voz-orb-wrap').classList.add('cargando');
    vozReproducirSonido('conectando');
    document.getElementById('voz-btn').classList.add('escuchando');

    navigator.mediaDevices.getUserMedia({ audio: { echoCancellation: true, noiseSuppression: true, autoGainControl: true } }).then(function(s){
      stream = s;
      ws = new WebSocket(WS_URL);
      ws.binaryType = 'arraybuffer';

      ws.onopen = function(){
        // Contexto de reproduccion a 24kHz (frecuencia nativa de Gemini Live)
        playCtx = new (window.AudioContext || window.webkitAudioContext)({ sampleRate: 24000 });
        resetPlayQueue();
        // Captura PCM a 16kHz
        micCtx = new (window.AudioContext || window.webkitAudioContext)({ sampleRate: 16000 });
        source = micCtx.createMediaStreamSource(stream);
        processor = micCtx.createScriptProcessor(4096, 1, 1);
        processor.onaudioprocess = function(e){
          if(!ws || ws.readyState !== 1) return;
          var float32 = e.inputBuffer.getChannelData(0);
          // Float32 -> Int16
          var int16 = new Int16Array(float32.length);
          for(var i=0;i<float32.length;i++){
            var s = Math.max(-1, Math.min(1, float32[i]));
            int16[i] = s < 0 ? s * 32768 : s * 32767;
          }
          // Int16 -> base64
          var bytes = new Uint8Array(int16.buffer);
          var bin = '';
          for(var j=0;j<bytes.length;j++) bin += String.fromCharCode(bytes[j]);
          ws.send(JSON.stringify({ tipo: 'audio', data: btoa(bin) }));
        };
        source.connect(processor);
        processor.connect(micCtx.destination);
      };

      ws.onmessage = function(e){
        try {
          var msg = JSON.parse(e.data);
          if(msg.tipo === 'audio' && msg.data) encolarPCM(msg.data, msg.sampleRate || 24000);
          if(msg.tipo === 'interrumpido') detenerReproduccionActual();
          if(msg.tipo === 'estado' && msg.texto) setEstado(msg.texto);
          if(msg.tipo === 'sesion_iniciada'){
            document.getElementById('voz-orb-wrap').classList.remove('cargando');
            setEstadoTemporal('¡Lista! Habla', 1400);
            vozReproducirSonido('listo');
          }
        } catch(ex){}
      };

      ws.onclose = function(){ setEstado('Desconectado'); vozDetener(); };
      ws.onerror = function(){ setEstado('Error de conexion'); vozDetener(); };

    }).catch(function(){
      setEstado('Sin acceso al microfono');
      activo = false;
      document.getElementById('voz-btn').classList.remove('escuchando');
    });
  }

  function vozDetener(){
    activo = false;
    document.getElementById('voz-orb-wrap').classList.remove('cargando');
    document.getElementById('voz-btn').classList.remove('escuchando');
    setEstado('Listo para hablar');
    resetPlayQueue();
    if(processor) { try { processor.disconnect(); } catch(ex){} processor = null; }
    if(source) { try { source.disconnect(); } catch(ex){} source = null; }
    if(micCtx) { try { micCtx.close(); } catch(ex){} micCtx = null; }
    if(playCtx) { try { playCtx.close(); } catch(ex){} playCtx = null; }
    if(stream) stream.getTracks().forEach(function(t){ t.stop(); }); stream = null;
    if(ws) { try { ws.close(); } catch(ex){} } ws = null;
  }
})();
</script>
